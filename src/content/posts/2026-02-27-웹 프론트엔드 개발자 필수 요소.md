### 3.2 TanStack Query 실전 패턴

```typescript
// 커스텀 훅으로 추상화
function useUser(userId: string) {
  return useQuery({
    queryKey: ['user', userId],
    queryFn: () => fetchUser(userId),
    staleTime: 5 * 60 * 1000, // 5분
    gcTime: 10 * 60 * 1000,   // 10분
  });
}

// Optimistic Update
function useUpdateUser() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: updateUser,
    onMutate: async (newUser) => {
      await queryClient.cancelQueries({ queryKey: ['user', newUser.id] });
      const previousUser = queryClient.getQueryData(['user', newUser.id]);
      queryClient.setQueryData(['user', newUser.id], newUser);
      return { previousUser };
    },
    onError: (err, newUser, context) => {
      queryClient.setQueryData(['user', newUser.id], context?.previousUser);
    },
  });
}
```

## 5) 웹 접근성 (Accessibility)

### 5.1 왜 접근성인가?

접근성은 단순한 윤리적 의무를 넘어 법적 요구사항이 되고 있습니다. 한국도 장애인차별금지법에 따라 웹 접근성 준수가 의무화되어 있습니다.

### 5.2 WCAG 2.2 핵심 원칙

- **인식 가능성 (Perceivable)**: 모든 정보는 사용자가 인식할 수 있어야 함
- **운용 가능성 (Operable)**: 모든 기능은 키보드로 조작 가능해야 함
- **이해 가능성 (Understandable)**: 콘텐츠와 UI는 이해하기 쉬워야 함
- **견고성 (Robust)**: 다양한 보조 기술과 호환되어야 함

### 5.3 실전 구현

```tsx
// 접근성 있는 모달
function AccessibleModal({ isOpen, onClose, title, children }: ModalProps) {
  return (
    <dialog
      aria-labelledby="modal-title"
      aria-modal="true"
      open={isOpen}
    >
      <h2 id="modal-title">{title}</h2>
      <div>{children}</div>
      <button onClick={onClose} aria-label="모달 닫기">✕</button>
    </dialog>
  );
}

// 접근성 있는 폼
function LoginForm() {
  const [error, setError] = useState('');

  return (
    <form>
      <label htmlFor="email">이메일</label>
      <input
        id="email"
        type="email"
        aria-required="true"
        aria-describedby={error ? 'email-error' : undefined}
        aria-invalid={!!error}
      />
      {error && (
        <p id="email-error" role="alert" aria-live="polite">
          {error}
        </p>
      )}
    </form>
  );
}
```

          onChange={handleInputChange}
          placeholder="메시지를 입력하세요..."
        />
        <button type="submit" disabled={isLoading}>전송</button>
      </form>
    </div>
  );
}
